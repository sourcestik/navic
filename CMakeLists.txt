cmake_minimum_required(VERSION 3.16)
project(navic VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
add_library(navic-core INTERFACE)
target_include_directories(navic-core INTERFACE backend/include backend/vendor)
add_executable(navic-api backend/src/main.cpp)
target_link_libraries(navic-api PRIVATE navic-core Threads::Threads CURL::libcurl)
add_executable(test-graph backend/tests/test_graph.cpp)
target_link_libraries(test-graph PRIVATE navic-core)
enable_testing()
add_test(NAME routing-correctness COMMAND test-graph)
